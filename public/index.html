Kotlin Data Classes (Retrofit / Moshi / Gson) 
data class AdsResponse(
    val scope: String? = null,
    val city: String? = null,
    val enabled: Boolean = true,
    val ads: List<AdItem> = emptyList()
)

data class AdItem(
    val id: String,
    val type: String, // "banner", "photo", "video"
    val title: String,
    val subtitle: String? = null,
    val description: String? = null,
    val mediaUrl: String,
    val actionUrl: String,
    val actionLabel: String,
    val sponsorTag: String? = null,
    val badgeText: String? = null,
    val isDismissible: Boolean = true,
    val locationInfo: String? = null,
    val rating: Double? = null
)
Retrofit API Interface 
interface AdsApiService {
    @GET("ads/global.json")
    suspend fun getGlobalAds(): AdsResponse

    @GET("ads/{city}.json")
    suspend fun getCityAds(@Path("city") city: String): AdsResponse
}

Admin Token & Cloudflare Settings

Token is automatically attached as x-admin-token header on all API calls.

Cloudflare Pages Setup:
1. Bind KV namespace as ADS_KV in Settings > Functions.
2. Set secret ADMIN_TOKEN in Settings > Environment variables.
